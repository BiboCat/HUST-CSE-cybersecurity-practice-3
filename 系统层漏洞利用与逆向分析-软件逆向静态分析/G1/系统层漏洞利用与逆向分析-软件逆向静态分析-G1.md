### 系统层漏洞利用与逆向分析-软件逆向静态分析-G1

#### 提示——字符串

题目提示我们使用字符串寻找线索，那我们就`Shift+F12`寻找一下。

发现`Enter the flag:`、`Wrong`、`Correct`等字段，说明这是一个flag校验程序。

最终追踪到了该函数，我做了一些标记：

```c
__int64 __fastcall core(__int64 a1, __int64 a2, int a3, int a4, int a5, int a6)
{
  __int64 length; // rdx
  __int64 check_flag_pointer; // rax
  __int64 result; // rax
  char flag[72]; // [rsp+0h] [rbp-58h] BYREF
  unsigned __int64 v10; // [rsp+48h] [rbp-10h]

  v10 = __readfsqword(0x28u);
  input_flag(2, (unsigned int)"Enter the flag: ", a3, a4, a5, a6, flag[0]);
  if ( sub_404E40(flag, 64LL, off_4AA6D8) )
  {
    flag[strcspn_(flag, "\n")] = 0;             // 我觉得它是发挥strcspn的功能，换行结尾改成\0结尾
    length = get_len(flag);
    check_flag_pointer = 0LL;
    if ( length == 42 )
    {
      while ( 1 )
      {
        length = (unsigned __int8)flag[check_flag_pointer] ^ 0xFFFFFFAA;// 这行不影响判断
        if ( ((unsigned __int8)flag[check_flag_pointer] ^ 0xAA) != true_flag[check_flag_pointer] )
          break;
        if ( ++check_flag_pointer == 42 )       // 上述循环使得check_flag_pointer顺利达到42
        {
          sub_4051D0("Correct!", "\n", length, true_flag);
          result = 0LL;
          goto LABEL_7;
        }
      }
    }
    sub_4051D0("Wrong!", "\n", length, true_flag);
    result = 1LL;
  }
  else
  {
    result = 1LL;
  }
LABEL_7:
  if ( v10 != __readfsqword(0x28u) )
    sub_41BC30();
  return result;
}
```

核心判断就一句：
`flag[check_flag_pointer] ^ 0xAA) == true_flag[check_flag_pointer]`

与`0xAA`异或后与真的flag对比，而`true_flag`的内容是程序硬编码的，点进去就能看到！

```
0CCh, 0C6h, 0CBh, 0CDh, 0D1h, 9Dh, 0CFh, 99h, 0CBh, 93h, 0CCh, 98h, 0C9h, 87h, 0C8h, 92h, 9Eh, 9Bh 87h, 9Eh, 0CEh, 9Ch, 9Dh, 87h, 0CBh, 9Ah, 0CFh, 9Fh, 87h, 9Bh, 0CCh, 9Ch, 0C9h, 92h, 0C8h, 99h, 93h 9Ah, 0CEh, 98h, 9Eh, 0D7h, 16h
```

好了那就编写一个脚本，把这一串的每一个字节，拿去和`0xAA`按位异或一下。

这种小事肯定是给AI跑啦！结果在`exp/xor_aa.py`，直接`python xor_aa.py`，全部都是硬编码。
