```c
static unsigned char rc4_crypt(const unsigned char *key, size_t key_len,
                               const unsigned char *src, size_t src_len,
                               unsigned char *dst)
{
    unsigned char S[256];
    unsigned char tmp, S_i, last_byte = 0;
    unsigned int  j;
    size_t        i;

    /* ---- KSA 第一步：S[i] = i ---- */
    for (i = 0; i < 256; ++i)
        S[i] = (unsigned char)i;

    /* ---- KSA 第二步：密钥调度 ---- */
    j = 0;
    for (i = 0; i < 256; ++i)
    {
        tmp = S[i];
        j  += (unsigned char)(key[i % key_len] + S[i]);
        last_byte = (unsigned char)j;          /* 死代码，见下 */
        S[i]                = S[(unsigned char)j];
        S[(unsigned char)j] = tmp;
    }

    /* ---- PRGA：生成密钥流并与 src 异或 ---- */
    if (src_len != 0)
    {
        j = 0;
        for (i = 1; i <= src_len; ++i)
        {
            S_i = S[i & 0xFF];
            j  += S_i;
            S[i & 0xFF] = S[j & 0xFF];
            S[j & 0xFF] = S_i;

            last_byte = (unsigned char)(src[i - 1] ^ S[(S[i & 0xFF] + S_i) & 0xFF]);
            dst[i - 1] = last_byte;
        }
    }
    return last_byte;
}
```

RC4是一种**对称流加密算法**，只要知道密钥，就能明文密文互相加密和解密！
而密钥，是硬编码在程序中的。

密钥：`2F 9E 4A C3 71 85 D6 1B`
