#### 提示——没有提示

从字符串视图可以得知，这个还是一个flag检测程序——追踪到核心函数后，让我们审计一下。

```c
__int64 core()
{
  FILE *v0; // rax
  char *flag_pointer; // rbx
  _BYTE *true_flag_pointer; // rsi
  char Buffer[42]; // [rsp+20h] [rbp-58h] BYREF
  char v5; // [rsp+4Ah] [rbp-2Eh] BYREF

  sub_1400016B0();
  echo_("Enter the flag: ");
  v0 = __acrt_iob_func(0);
  if ( !fgets(Buffer, 64, v0) )
    return 1i64;
  Buffer[strcspn(Buffer, "\n")] = 0;
  if ( strlen(Buffer) == 42 )
  {
    flag_pointer = Buffer;
    for ( true_flag_pointer = &true_flag;
          (unsigned __int8)oprate_func((unsigned __int8)*flag_pointer) == *true_flag_pointer;// 会先对flag中的字符进行处理
          ++true_flag_pointer )
    {
      if ( ++flag_pointer == &v5 )              // flag_pointer是指针，指着buffer栈上移动，最后移动到v5就是读取完整个flag
      {
        puts("Correct!");
        return 0i64;
      }
    }
  }
  puts("Wrong!");
  return 1i64;
}

__int64 __fastcall oprate_func(unsigned __int8 the_char)
{
  unsigned __int8 v1; // al
  unsigned __int8 v2; // al

  v1 = ROL1_4(the_char);                        // ROL是循环左移，ROL1表示对1个字节的循环左移
  v2 = ROR1_3(v1);                              // 一下左移动4一下右移动3，那等于循环左移1位
  return final_opration(&S_box, v2);
}

__int64 __fastcall final_opration(__int64 S_box, unsigned __int8 the_char)
{
  __int64 result; // rax

  result = 16 * (unsigned int)*(unsigned __int8 *)(S_box + (the_char >> 4));// >>是右移4位，不循环，相当于取高半字节
  LOBYTE(result) = *(_BYTE *)(S_box + (the_char & 0xF)) | (16 * *(_BYTE *)(S_box + (the_char >> 4)));// LOBYTE取最低8字节；& 0xF与1111相与，取低半字节
                                                // 所以是高半字节做偏移选S_box值，低半字节也选一个S_box值——再注意S_box值的每个字节都只有4位（0-15）
                                                // 高半选的S_box值乘以16，放前4字节，后4字节为0；低半选的放后4字节，前4字节为0——因此这个 | 相当于拼接
  return result;
}
```

上面的代码注释已经把处理流程讲清楚了，特别是`final_opration`这个函数，用到了一个S盒：对输入的字节，分成高低两半字节，分别作为S盒索引选出2个4比特的半字节；高半字节选出的半字节作为`result`的高半字节，低半字节选出的半字节作为`result`的低半字节。

**想要逆向这个函数也很简单，如果S盒是一个双射映射，就可以反推**！

S盒值如下：`7 D A 4 F 1 B 9 2 E 6 0 C 8 3 5`   ——很明显这是一个双射！

那么可以推断出**逆S盒**为`B 5 8 E 3 F A 0 D 7 2 6 C 1 9 4`。

逆推完`final_opration`再逆推一下`ROL1_4`和`ROR1_3`，它们原本的结果是循环左移1位，那么反过来我们**循环右移1位**即可！
